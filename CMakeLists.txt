cmake_minimum_required(VERSION 3.22)
project(kmm_eastmoney_price VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network Widgets)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6XmlGui REQUIRED)
find_package(LibAlkimia6 CONFIG QUIET)
if(TARGET Alkimia::alkimia)
  set(ALKIMIA_LIBRARY Alkimia::alkimia)
else()
  # Arch's libalkimia6.pc contains a relative include path.
  find_path(ALKIMIA_INCLUDE_DIR alkimia/alkvalue.h PATH_SUFFIXES alkimia/Qt6 REQUIRED)
  find_library(ALKIMIA_LIBRARY alkimia6 REQUIRED)
endif()
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_path(KMM_INCLUDE_DIR kmymoneyplugin.h PATH_SUFFIXES kmymoney REQUIRED)
find_library(KMM_PLUGIN_LIBRARY kmm_plugin REQUIRED)
find_library(KMM_MYMONEY_LIBRARY kmm_mymoney REQUIRED)

add_library(quote_core STATIC src/quotes.cpp src/quoteclient.cpp)
target_include_directories(quote_core PUBLIC src ${KMM_INCLUDE_DIR} ${ALKIMIA_INCLUDE_DIR})
target_link_libraries(quote_core PUBLIC Qt6::Core Qt6::Network ${ALKIMIA_LIBRARY} ${GMPXX_LIBRARY} ${GMP_LIBRARY} ${KMM_MYMONEY_LIBRARY})
target_compile_options(quote_core PRIVATE -Wall -Wextra -Wpedantic)

add_library(ledger_writer STATIC src/ledgerwriter.cpp)
target_link_libraries(ledger_writer PUBLIC quote_core)
target_include_directories(ledger_writer PRIVATE compat)
target_compile_options(ledger_writer PRIVATE -Wall -Wextra -Wpedantic)
add_library(kmm_eastmoney_price MODULE src/plugin.cpp src/resources.qrc)
set_target_properties(kmm_eastmoney_price PROPERTIES PREFIX "")
target_link_libraries(kmm_eastmoney_price PRIVATE ledger_writer Qt6::Widgets KF6::CoreAddons KF6::ConfigCore KF6::XmlGui ${KMM_PLUGIN_LIBRARY})
target_compile_options(kmm_eastmoney_price PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(kmm_eastmoney_price PRIVATE compat)
set(KMM_PLUGIN_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kmymoney_plugins" CACHE PATH "KMyMoney plugin directory, relative to prefix or absolute")
install(TARGETS kmm_eastmoney_price LIBRARY DESTINATION ${KMM_PLUGIN_INSTALL_DIR})

add_executable(quote-smoke tools/smoke.cpp)
target_link_libraries(quote-smoke PRIVATE quote_core)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test_quotes tests/test_quotes.cpp)
  target_link_libraries(test_quotes PRIVATE quote_core Qt6::Test)
  add_test(NAME quotes COMMAND test_quotes)
  add_executable(test_network tests/test_network.cpp)
  target_link_libraries(test_network PRIVATE quote_core Qt6::Test)
  add_test(NAME network COMMAND test_network)
  add_executable(test_plugin tests/test_plugin.cpp)
  target_include_directories(test_plugin PRIVATE compat)
  target_link_libraries(test_plugin PRIVATE quote_core Qt6::Widgets Qt6::Test KF6::CoreAddons KF6::XmlGui ${KMM_PLUGIN_LIBRARY})
  add_test(NAME plugin_metadata COMMAND test_plugin $<TARGET_FILE:kmm_eastmoney_price>)
  set_tests_properties(plugin_metadata PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME plugin_discovery COMMAND test_plugin --discover-install ${CMAKE_BINARY_DIR} ${KMM_PLUGIN_INSTALL_DIR} ${CMAKE_COMMAND})
  set_tests_properties(plugin_discovery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(test_ledger tests/test_ledger.cpp)
  target_include_directories(test_ledger PRIVATE compat)
  target_link_libraries(test_ledger PRIVATE ledger_writer Qt6::Widgets Qt6::Test)
  add_test(NAME ledger COMMAND test_ledger)
  set_tests_properties(ledger PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
